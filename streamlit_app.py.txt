@st.cache_resource(show_spinner=False)
def load_excel_file(file_bytes: bytes):
    """
    Carga el archivo Excel en memoria y retorna un objeto pd.ExcelFile.
    Se usa st.cache_resource porque pd.ExcelFile NO es serializable
    y produce errores con st.cache_data.
    """
    try:
        excel_obj = pd.ExcelFile(io.BytesIO(file_bytes))
        return excel_obj
    except Exception as e:
        st.error(f"❌ Error al leer el archivo Excel: {e}")
        return None


@st.cache_data(show_spinner=False)
def parse_sheet_to_df(file_bytes: bytes, sheet_name: str) -> pd.DataFrame:
    """
    Convierte una hoja específica del Excel en un DataFrame de pandas.
    Se cachea con st.cache_data porque recibe/retorna objetos serializables
    (bytes y DataFrame), evitando el error de hashing con pd.ExcelFile.
    """
    try:
        df = pd.read_excel(io.BytesIO(file_bytes), sheet_name=sheet_name)
        df = clean_dataframe(df)
        return df
    except Exception as e:
        st.error(f"❌ Error al convertir la hoja '{sheet_name}' en DataFrame: {e}")
        return pd.DataFrame()


def clean_dataframe(df: pd.DataFrame) -> pd.DataFrame:
    """
    Limpieza básica y robusta del DataFrame:
    - Elimina espacios en nombres de columnas.
    - Intenta convertir columnas con nombres relacionados a fecha.
    - Normaliza strings de columnas categóricas comunes.
    """
    df = df.copy()
    df.columns = [str(c).strip() for c in df.columns]

    # Intentar detectar y convertir columnas de fecha por nombre
    posibles_fechas = [c for c in df.columns if any(
        k in c.lower() for k in ["fecha", "date", "dia", "día"]
    )]
    for col in posibles_fechas:
        try:
            df[col] = pd.to_datetime(df[col], errors="coerce")
        except Exception:
            pass

    # Normalizar texto en columnas tipo "genero"/"ciudad" si existen
    for col in df.columns:
        if df[col].dtype == object:
            df[col] = df[col].astype(str).str.strip()

    return df


@st.cache_data(show_spinner=False)
def generate_synthetic_data(n: int = 500) -> pd.DataFrame:
    """
    Genera un dataset sintético de demostración con columnas típicas:
    Fecha, Ciudad, Género, Categoría, Ventas, Edad, Satisfacción.
    """
    np.random.seed(42)
    ciudades = ["Bogotá", "Medellín", "Cali", "Barranquilla", "Cartagena",
                "Bucaramanga", "Pereira", "Manizales"]
    generos = ["Masculino", "Femenino"]
    categorias = ["Electrónica", "Hogar", "Moda", "Deportes", "Alimentos"]

    fechas = pd.date_range(start="2024-01-01", end="2024-12-31", periods=n)

    df = pd.DataFrame({
        "Fecha": np.random.choice(fechas, n),
        "Ciudad": np.random.choice(ciudades, n, p=[0.25, 0.18, 0.15, 0.12,
                                                     0.10, 0.08, 0.07, 0.05]),
        "Género": np.random.choice(generos, n, p=[0.52, 0.48]),
        "Categoría": np.random.choice(categorias, n),
        "Ventas": np.round(np.random.gamma(5, 150, n), 2),
        "Edad": np.random.randint(18, 70, n),
        "Satisfacción": np.round(np.random.uniform(1, 5, n), 1),
    })
    df = df.sort_values("Fecha").reset_index(drop=True)
    return df


def detect_column_types(df: pd.DataFrame) -> dict:
    """
    Detecta y clasifica las columnas del DataFrame según su tipo:
    fechas, numéricas, categóricas (incluye detección heurística
    de columnas de 'género' y 'ciudad').
    """
    tipos = {
        "fecha": [],
        "numerica": [],
        "categorica": [],
        "genero": None,
        "ciudad": None,
    }

    for col in df.columns:
        if pd.api.types.is_datetime64_any_dtype(df[col]):
            tipos["fecha"].append(col)
        elif pd.api.types.is_numeric_dtype(df[col]):
            tipos["numerica"].append(col)
        else:
            tipos["categorica"].append(col)

    # Heurística para encontrar columna de género
    for col in tipos["categorica"]:
        if any(k in col.lower() for k in ["genero", "género", "sexo", "gender"]):
            tipos["genero"] = col
            break

    # Heurística para encontrar columna de ciudad
    for col in tipos["categorica"]:
        if any(k in col.lower() for k in ["ciudad", "city", "municipio", "localidad"]):
            tipos["ciudad"] = col
            break

    return tipos

        df_tendencia = df_tendencia.sort_values(fecha_sel)
        df_agrupado = df_tendencia.groupby(
            df_tendencia[fecha_sel].dt.to_period("D").apply(lambda r: r.start_time)
        )[metrica_sel].sum().reset_index()
        df_agrupado.columns = [fecha_sel, metrica_sel]

        fig_linea = px.line(
            df_agrupado, x=fecha_sel, y=metrica_sel,
            markers=True,
            title=f"Evolución de {metrica_sel} en el tiempo",
            color_discrete_sequence=["#00B4D8"]
        )
        fig_linea.update_traces(line=dict(width=3), marker=dict(size=6, color="#90E0EF"))
        fig_linea.update_layout(
            paper_bgcolor="#172A45",
            plot_bgcolor="#172A45",
            font_color="#E6F1FF",
            title_font_color="#90E0EF",
            xaxis=dict(gridcolor="#263B55"),
            yaxis=dict(gridcolor="#263B55")
        )
        st.plotly_chart(fig_linea, use_container_width=True)
    else:
        st.info("ℹ️ Se requiere al menos una columna de fecha y una columna numérica para este gráfico.")

# ---------- TAB 2: Comparativo por Categoría ----------
with tab2:
    categoricas_disponibles = col_types["categorica"]
    if categoricas_disponibles and numeric_cols:
        cat_sel = st.selectbox(
            "Selecciona la variable categórica", categoricas_disponibles, key="cat_barras"
        )
        metrica_barras = st.selectbox(
            "Selecciona la métrica numérica", numeric_cols, key="metrica_barras"
        )

        df_barras = df_filtrado.groupby(cat_sel)[metrica_barras].sum().reset_index()
        df_barras = df_barras.sort_values(metrica_barras, ascending=False)

        fig_barras = px.bar(
            df_barras, x=cat_sel, y=metrica_barras,
            color=cat_sel,
            title=f"{metrica_barras} por {cat_sel}",
            color_discrete_sequence=px.colors.sequential.Blues_r
        )
        fig_barras.update_layout(
            paper_bgcolor="#172A45",
            plot_bgcolor="#172A45",
            font_color="#E6F1FF",
            title_font_color="#90E0EF",
            showlegend=False,
            xaxis=dict(gridcolor="#263B55"),
            yaxis=dict(gridcolor="#263B55")
        )
        st.plotly_chart(fig_barras, use_container_width=True)
    else:
        st.info("ℹ️ Se requiere al menos una columna categórica y una numérica para este gráfico.")

# ---------- TAB 3: Composición Porcentual (Dona) ----------
with tab3:
    if categoricas_disponibles:
        cat_dona = st.selectbox(
            "Selecciona la variable categórica", categoricas_disponibles, key="cat_dona"
        )
        conteo_dona = df_filtrado[cat_dona].value_counts().reset_index()
        conteo_dona.columns = [cat_dona, "Cantidad"]

        fig_dona = px.pie(
            conteo_dona, names=cat_dona, values="Cantidad",

            plot_bgcolor="#172A45",
            font_color="#E6F1FF",
            title_font_color="#90E0EF",
            legend_font_color="#E6F1FF"
        )
        st.plotly_chart(fig_dona, use_container_width=True)
